{%- comment -%} A picture for a story or edition. Uses the photo named in include.photo (an id in _data/photos.yml), else a library photo tagged with the category include.cat (rotating by include.n), else the category's illustration. Params: photo, cat (category object), n (number), class, eager (true for the first big image), caption (true to show credit). {%- endcomment -%} {%- assign ph = nil -%} {%- if include.photo -%}{%- assign ph = site.data.photos | where: "id", include.photo | first -%}{%- endif -%} {%- unless used_photos -%}{%- assign used_photos = "|" -%}{%- endunless -%} {%- unless ph -%} {%- comment -%} A library photo for the category that this page has not shown yet, so pictures don't repeat. {%- endcomment -%} {%- assign cid = include.cat.id -%} {%- assign pool = site.data.photos | where_exp: "p", "p.categories contains cid" -%} {%- if pool.size > 0 -%} {%- assign start = include.n | default: 0 | modulo: pool.size -%} {%- for k in (1..pool.size) -%} {%- assign i = start | plus: k | minus: 1 | modulo: pool.size -%}{%- assign cand = pool[i] -%} {%- capture key -%}|{{ cand.id }}|{%- endcapture -%} {%- unless used_photos contains key -%}{%- assign ph = cand -%}{%- break -%}{%- endunless -%} {%- endfor -%} {%- endif -%} {%- endunless -%} {%- if ph -%}{%- assign used_photos = used_photos | append: ph.id | append: "|" -%}{%- endif -%} {%- if ph -%} {%- else -%} {%- endif -%}
